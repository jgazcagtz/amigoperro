// Simple Firestore Security Rules for Development
// Copy these rules to your Firebase Console > Firestore Database > Rules
// WARNING: These rules allow full access - only use for development/testing

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow all authenticated users to read pending walks (so paseadores can see available paseos)
    match /walks/{walkId} {
      allow read: if request.auth != null && 
        (resource.data.status == 'pending' || true);
      allow write: if request.auth != null;
    }
    
    // Allow full access to all other collections for development
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
